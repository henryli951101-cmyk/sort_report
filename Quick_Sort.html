void quickSort(int arr[], int low, int high) { if (low < high) { int pivot = arr[high]; // 選最後一個當基準 int i = (low - 1); for (int j = low; j <= high - 1; j++) { if (arr[j] < pivot) { i++; int temp = arr[i]; arr[i] = arr[j]; arr[j] = temp; } } int temp = arr[i + 1]; arr[i + 1] = arr[high]; arr[high] = temp; int pi = i + 1; // 分割點 quickSort(arr, low, pi - 1); quickSort(arr, pi + 1, high); } }
